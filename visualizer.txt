import heapq
from typing import Any, List, Dict, Optional, Set, Tuple
from graph import Graph
from drone import Drone


class Scheduler:
    """Air Traffic Control: Manages drone movement, capacities, and turns."""

    def __init__(
        self,
        graph: Graph,
        drones: List[Drone],
        start_hub: str,
        end_hub: str,
    ) -> None:
        self.graph = graph
        self.drones = drones
        self.start_hub = start_hub
        self.end_hub = end_hub
        self.turn_number = 0

        # Room capacities: How many drones are currently in each zone?
        self.room_occupancy: Dict[str, int] = {
            node: 0 for node in graph.nodes
        }
        self.room_occupancy[self.start_hub] = len(self.drones)

        # Road capacities: How many drones are currently on a connection?
        self.connection_occupancy: Dict[str, int] = {}

        # Visualizer frame history (Records Turn 0 at start_hub!)
        self.history: List[Dict[str, Any]] = []
        self._record_snapshot([])

    def _record_snapshot(self, turn_movements: List[str]) -> None:
        """Captures a snapshot for the visualizer."""
        snapshot: Dict[str, Any] = {
            "turn": self.turn_number,
            "events": turn_movements.copy(),
            "drones": {},
        }
        for d in self.drones:
            snapshot["drones"][d.id] = {
                "location": d.current_location,
                "transit_target": d.transit_target,
                "is_in_transit": d.turns_until_arrival > 0,
                "is_delivered": d.is_delivered,
            }
        self.history.append(snapshot)

    def get_connection_key(self, zone_a: str, zone_b: str) -> str:
        """Forces alphabetical order to prevent directional overlap bugs."""
        first = min(zone_a, zone_b)
        second = max(zone_a, zone_b)
        return f"{first}-{second}"

    def get_max_room_capacity(self, zone: str) -> float:
        """Default max_drones is 1 for hubs but unlimited for start and end."""
        if zone in (self.start_hub, self.end_hub):
            return float("inf")
        return float(self.graph.nodes[zone].get("max_drones", 1))

    def get_max_link_capacity(self, zone_a: str, zone_b: str) -> float:
        """Finds max_link_capacity between two zones (Default is 1)."""
        for neighbor, metadata in self.graph.get_neighbor(zone_a):
            if neighbor == zone_b:
                return float(metadata.get("max_link_capacity", 1))
        return 1.0

    def _can_enter(self, current_zone: str, next_zone: str) -> bool:
        """Checks if both the target zone and connecting link have capacity."""
        max_room = self.get_max_room_capacity(next_zone)
        max_link = self.get_max_link_capacity(current_zone, next_zone)
        conn_key = self.get_connection_key(current_zone, next_zone)

        current_room_load = self.room_occupancy[next_zone]
        current_link_load = self.connection_occupancy.get(conn_key, 0)
        return current_room_load < max_room and current_link_load < max_link

    def _get_zone_weight(self, zone: str) -> float:
        """Returns pathfinding weight:restricted=2.0,priority=0.9,normal=1.0"""
        zone_type = self.graph.nodes[zone].get("zone", "normal")
        if zone_type == "restricted":
            return 2.0
        if zone_type == "priority":
            return 0.9
        return 1.0

    def _calculate_path_cost(self, path: List[str]) -> float:
        """Calculates the remaining weighted turn cost of a path list."""
        cost = 0.0
        for zone in path:
            cost += self._get_zone_weight(zone)
        return cost

    def _find_open_detour(
        self, current_zone: str, max_acceptable_cost: float
    ) -> Optional[List[str]]:
        """Finds an alternative path whose immediate first step is open."""
        priority_queue: List[Tuple[float, str, List[str]]] = [
            (0.0, current_zone, [current_zone])
        ]
        distances: Dict[str, float] = {current_zone: 0.0}

        while priority_queue:
            current_cost, node, path = heapq.heappop(priority_queue)

            if current_cost > max_acceptable_cost:
                continue

            if node == self.end_hub and len(path) > 1:
                return path[1:]

            if current_cost > distances.get(node, float("inf")):
                continue

            for neighbor, _ in self.graph.get_neighbor(node):
                neighbor_type = self.graph.nodes[neighbor].get(
                    "zone", "normal"
                )
                if neighbor_type == "blocked":
                    continue

                # The immediate first step from current_zone must be open
                if (
                    node == current_zone
                    and not self._can_enter(current_zone, neighbor)
                ):
                    continue

                step_cost = self._get_zone_weight(neighbor)
                new_cost = current_cost + step_cost

                if (
                    new_cost <= max_acceptable_cost
                    and new_cost < distances.get(neighbor, float("inf"))
                ):
                    distances[neighbor] = new_cost
                    heapq.heappush(
                        priority_queue,
                        (new_cost, neighbor, path + [neighbor]),
                    )

        return None

    def run_simulation(self) -> None:
        """The master loop. Runs until every drone is delivered."""
        while not all(drone.is_delivered for drone in self.drones):
            self.turn_number += 1
            self.play_turn()

    def play_turn(self) -> None:
        """Executes a single tick of the simulation."""
        turn_output: List[str] = []
        has_moved_this_turn: Set[int] = set()
        normal_links_used: List[str] = []

        # ==========================================
        # PHASE 1: PROCESS ARRIVALS (Mid-Air Drones)
        # ==========================================
        for drone in self.drones:
            if drone.turns_until_arrival > 0:
                drone.turns_until_arrival -= 1
                # Lock this drone so Phase 2 cannot move it again
                has_moved_this_turn.add(drone.id)

                if (
                    drone.turns_until_arrival == 0
                    and drone.transit_target is not None
                ):
                    conn_key = self.get_connection_key(
                        drone.current_location, drone.transit_target
                    )

                    # 1. Free up the road it was hovering on
                    self.connection_occupancy[conn_key] -= 1

                    # 2. Officially update its location
                    drone.current_location = drone.transit_target
                    drone.transit_target = None

                    # 3. Log the arrival at the destination
                    turn_output.append(f"D{drone.id}-{drone.current_location}")

                    if drone.current_location == self.end_hub:
                        drone.is_delivered = True
                        drone.delivery_turn = self.turn_number
                        self.room_occupancy[self.end_hub] -= 1

        # ==========================================
        # PHASE 2: ATTEMPT TO MOVE AVAILABLE DRONES
        # ==========================================
        for drone in self.drones:
            if (
                drone.id in has_moved_this_turn
                or drone.is_delivered
                or drone.turns_until_arrival > 0
                or not drone.path
            ):
                continue

            next_zone = drone.path[0]

            # If the primary next step is full check
            # for an open parallel detour
            if not self._can_enter(drone.current_location, next_zone):
                remaining_cost = self._calculate_path_cost(drone.path)
                detour = self._find_open_detour(
                    drone.current_location, remaining_cost + 2.0
                )
                if detour:
                    drone.path = detour
                    next_zone = drone.path[0]

            # ==========================================
            # PHASE 3: THE BOUNCER (Check Permissions)
            # ==========================================
            if self._can_enter(drone.current_location, next_zone):
                conn_key = self.get_connection_key(
                    drone.current_location, next_zone
                )
                current_link_load = self.connection_occupancy.get(conn_key, 0)

                self.room_occupancy[drone.current_location] -= 1
                self.room_occupancy[next_zone] += 1
                self.connection_occupancy[conn_key] = current_link_load + 1

                zone_type = self.graph.nodes[next_zone].get("zone", "normal")

                if zone_type == "restricted":
                    drone.transit_target = next_zone
                    drone.turns_until_arrival = 1
                    directional_conn = f"{drone.current_location}-{next_zone}"
                    turn_output.append(f"D{drone.id}-{directional_conn}")
                else:
                    drone.current_location = next_zone
                    normal_links_used.append(conn_key)
                    turn_output.append(f"D{drone.id}-{next_zone}")

                    if next_zone == self.end_hub:
                        drone.is_delivered = True
                        drone.delivery_turn = self.turn_number
                        self.room_occupancy[self.end_hub] -= 1

                drone.path.pop(0)

        # ==========================================
        # PHASE 4: PRINT TURN RESULTS & RECORD FRAME
        # ==========================================
        if turn_output:
            turn_output.sort(key=lambda x: int(x.split("-")[0][1:]))
            print(" ".join(turn_output))

        for conn_key in normal_links_used:
            self.connection_occupancy[conn_key] -= 1

        self._record_snapshot(turn_output)
